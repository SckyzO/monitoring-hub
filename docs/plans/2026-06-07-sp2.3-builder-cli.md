# SP2.3 — Distribution builder + `mh repo build` Implementation Plan

> **For agentic workers:** executed inline (superpowers:executing-plans), TDD per step.
> Container-first: all tooling via `./dev run <cmd>`; `make ci` on host. Steps use
> checkbox (`- [ ]`) syntax.

**Goal:** Assemble the full distribution trees — the Pages metadata tree
(`./public`) and the Releases staging tree (`./release`) — from built packages +
a catalog, populate each `Artifact.url`, and expose it as `mh repo build`.

**Architecture:** `repo/builder.py` orchestrates the SP2.1/SP2.2 primitives
(`build_rpm_repo`, `build_apt_repo`, `sign_repomd`, `sign_apt_release`) over the
catalog. It is the only new logic; rpm/deb generation and signing already exist.
A new `mh repo build` Click command wires it behind the injected `SubprocessRunner`.

**Tech Stack:** Python 3.12, pydantic v2, click, pytest+cov (≥95%, target 100%),
the SP1 `FakeRunner` (extended locally to materialize sentinel output files).

---

## Locked design decisions

1. **RPM release-tag scheme = per-(target, rpm_arch)** (validated with user;
   supersedes spec §2.1 `rpm-<name>-<version>`). One `repodata/` per
   `(target, rpm_arch)` ⇒ one `createrepo_c --location-prefix` ⇒ one tag
   `rpm-<target>-<rpm_arch>`. Consistent with the APT side (`apt-<codename>`).
   The spec is updated in Task 1.
2. **APT release-tag scheme = per-codename** (`apt-<codename>`, spec §2.1, already
   built in SP2.2). Codename derived **structurally** from the deb artifact's
   `target` (filenames don't encode it).
3. **GOARCH → rpm arch translation**: `amd64→x86_64`, `arm64→aarch64`. Artifact
   `arch` is GOARCH (`amd64`/`arm64`, manifest `build.archs`); the rpm *filename*
   and the Pages path use the rpm arch. DEB filenames keep GOARCH (`amd64`).
4. **Codename map** (from `DEFAULT_DEB_TARGETS` + ubuntu-26.04 "Resolute Raccoon"):
   `ubuntu-22.04→jammy, ubuntu-24.04→noble, ubuntu-26.04→resolute,
   debian-12→bookworm, debian-13→trixie`. Unknown target → `DistributionError`.
5. **`pages_base_url` param added** to `build_distribution` (spec §5.4 omitted it
   but mandates dashboards → Pages URL). Dashboard URL =
   `<pages_base_url>/dashboards/<name>.json`.
6. **Docker artifacts**: `Artifact.url` left untouched (populated at publish, SP2.4).
7. **`packages_dir` layout**: the existing `mh build` work tree
   `<fmt>/<target>/<arch>/*.<fmt>`; the builder locates a package by reconstructing
   its filename and `rglob`-ing for it (robust to the exact subdir depth).
8. **catalog.json**: the builder *writes* the URL-populated catalog to
   `public_out/catalog.json` (it does not copy a source file).
9. **GPG public key**: optional `public_key: Path | None`; copied to
   `public_out/RPM-GPG-KEY-monitoring-hub` + `public_out/apt/monitoring-hub.asc`
   when provided (key *export* is CI-side). `None` in dev/tests.

```python
"""Distribution builder (spec §5.4): assemble ./public + ./release from a catalog.

Orchestrates the SP2.1/SP2.2 primitives over the built packages, signs metadata
when a key is given, copies dashboards + the public key, writes the URL-populated
catalog.json to the Pages tree, and returns the updated Catalog.
"""
from __future__ import annotations

import shutil
from pathlib import Path

from forge.catalog.builder import write_catalog
from forge.domain.artifact import Artifact
from forge.domain.catalog import Catalog
from forge.domain.errors import DistributionError
from forge.packaging.runner import CommandRunner
from forge.repo.deb import build_apt_repo
from forge.repo.metadata_sign import sign_apt_release, sign_repomd
from forge.repo.rpm import build_rpm_repo

_ORIGIN = "monitoring-hub"
_RPM_ARCH = {"amd64": "x86_64", "arm64": "aarch64"}
_DEB_CODENAMES = {
    "ubuntu-22.04": "jammy",
    "ubuntu-24.04": "noble",
    "ubuntu-26.04": "resolute",
    "debian-12": "bookworm",
    "debian-13": "trixie",
}


def codename_for(target: str) -> str:
    """Map a deb target (ubuntu-24.04) to its apt codename (noble)."""
    try:
        return _DEB_CODENAMES[target]
    except KeyError as exc:
        raise DistributionError(f"no apt codename for deb target {target!r}") from exc


def _rpm_arch(arch: str | None) -> str:
    if arch not in _RPM_ARCH:
        raise DistributionError(f"no rpm arch mapping for {arch!r}")
    return _RPM_ARCH[arch]


def _rpm_filename(name: str, version: str, target: str, rpm_arch: str) -> str:
    return f"{name}-{version}-1.{target}.{rpm_arch}.rpm"


def _deb_filename(name: str, version: str, arch: str) -> str:
    return f"{name.replace('_', '-')}_{version}-1_{arch}.deb"


def artifact_hosted_url(
    *, name: str, version: str, artifact: Artifact, package_base_url: str, pages_base_url: str
) -> str | None:
    """Hosted URL for one artifact, or None for kinds populated elsewhere (docker)."""
    if artifact.type == "rpm":
        rpm_arch = _rpm_arch(artifact.arch)
        fn = _rpm_filename(name, version, str(artifact.target), rpm_arch)
        return f"{package_base_url}/rpm-{artifact.target}-{rpm_arch}/{fn}"
    if artifact.type == "deb":
        codename = codename_for(str(artifact.target))
        fn = _deb_filename(name, version, str(artifact.arch))
        return f"{package_base_url}/apt-{codename}/{fn}"
    if artifact.type == "grafana-dashboard":
        return f"{pages_base_url}/dashboards/{name}.json"
    return None
```

class SentinelRunner:
    """Fake runner: emulates createrepo_c / apt-ftparchive / gpg side effects."""
    def __init__(self) -> None:
        self.calls: list[list[str]] = []
    def run(self, args, *, cwd=None, env=None, stdin=None):
        self.calls.append(list(args))
        exe = args[0]
        if exe == "createrepo_c":
            work = Path(args[-1]); (work / "repodata").mkdir(parents=True, exist_ok=True)
            (work / "repodata" / "repomd.xml").write_text("<repomd/>")
            return CommandResult(args=list(args), returncode=0, stdout="", stderr="")
        if exe == "apt-ftparchive":
            out = "Package: x\n" if args[1] == "packages" else "Origin: monitoring-hub\n"
            return CommandResult(args=list(args), returncode=0, stdout=out, stderr="")
        if exe == "gpg":
            i = args.index("--output"); Path(args[i + 1]).write_text("sig")

```python
def _locate(packages_dir: Path, filename: str) -> Path:
    found = next(packages_dir.rglob(filename), None)
    if found is None:
        raise DistributionError(f"built package not found: {filename}")
    return found


def build_distribution(  # noqa: PLR0913 — orchestrator with explicit I/O params
    *,
    catalog: Catalog,
    packages_dir: Path,
    dashboards_dir: Path,
    public_out: Path,
    release_out: Path,
    package_base_url: str,
    pages_base_url: str,
    key_id: str | None = None,
    public_key: Path | None = None,
    runner: CommandRunner,
) -> Catalog:
    """Assemble the Pages (./public) and Releases (./release) trees from a catalog."""
    rpm_groups: dict[tuple[str, str], list[Path]] = {}
    deb_groups: dict[str, list[Path]] = {}
    updated_items = []

    for item in catalog.items:
        new_artifacts = []
        for art in item.artifacts:
            url = artifact_hosted_url(
                name=item.name, version=item.version, artifact=art,
                package_base_url=package_base_url, pages_base_url=pages_base_url,
            )
            if art.type == "rpm":
                rpm_arch = _rpm_arch(art.arch)
                fn = _rpm_filename(item.name, item.version, str(art.target), rpm_arch)
                tag_dir = release_out / f"rpm-{art.target}-{rpm_arch}"
                tag_dir.mkdir(parents=True, exist_ok=True)
                staged = Path(shutil.copy2(_locate(packages_dir, fn), tag_dir / fn))
                rpm_groups.setdefault((str(art.target), rpm_arch), []).append(staged)
            elif art.type == "deb":
                fn = _deb_filename(item.name, item.version, str(art.arch))
                deb_groups.setdefault(codename_for(str(art.target)), []).append(
                    _locate(packages_dir, fn)
                )
            elif art.type == "grafana-dashboard":
                src = dashboards_dir / f"{item.name}.json"
                if not src.is_file():
                    raise DistributionError(f"dashboard json not found: {src}")
                dash_dir = public_out / "dashboards"
                dash_dir.mkdir(parents=True, exist_ok=True)
                shutil.copy2(src, dash_dir / src.name)
            new_artifacts.append(art.model_copy(update={"url": url}) if url else art)
        updated_items.append(item.model_copy(update={"artifacts": new_artifacts}))

    for (target, rpm_arch), pkgs in rpm_groups.items():
        repodata_dir = public_out / target / rpm_arch / "repodata"
        build_rpm_repo(
            packages=pkgs, repodata_dir=repodata_dir,
            location_prefix=f"{package_base_url}/rpm-{target}-{rpm_arch}/", runner=runner,
        )
        if key_id is not None:
            sign_repomd(repodata_dir / "repomd.xml", key_id=key_id, runner=runner)

    for codename, pkgs in deb_groups.items():
        repo_dir = release_out / f"apt-{codename}"
        build_apt_repo(
            packages=pkgs, repo_dir=repo_dir, codename=codename, origin=_ORIGIN, runner=runner,
        )
        if key_id is not None:
            sign_apt_release(repo_dir / "Release", key_id=key_id, runner=runner)

    if public_key is not None:
        public_out.mkdir(parents=True, exist_ok=True)
        shutil.copy2(public_key, public_out / "RPM-GPG-KEY-monitoring-hub")
        apt_dir = public_out / "apt"
        apt_dir.mkdir(parents=True, exist_ok=True)
        shutil.copy2(public_key, apt_dir / "monitoring-hub.asc")

    updated = catalog.model_copy(update={"items": updated_items})
    write_catalog(updated, public_out / "catalog.json")
    return updated
```
