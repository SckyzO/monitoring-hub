# SP1.5 — Catalog builder + CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to
> implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the kind-agnostic `catalog.json` builder and the full `mh` CLI
surface (`validate`, `pull`, `build`, `catalog build`, `list`), then migrate the
34 legacy exporter manifests into the new `catalog/exporters/` data tree.

**Architecture:** `catalog/builder.py` assembles `CatalogEntry` objects into a
`Catalog` envelope (spec §10), diffing against a previous `catalog.json` to
compute `new`/`updated`. The CLI is a thin Click orchestration over the existing
`sources.resolver`, `kinds.registry`, and `catalog.builder` — strictly
kind-agnostic (never exporter-specific): it goes through `get_producer(kind)` so
a second kind plugs in with zero CLI change. The 34 legacy manifests are
transformed by a throwaway script (inject `kind: exporter`, wrap
`upstream`/`build`/`artifacts` under `spec`, drop `new`/`updated`), validated via
the freshly-built `mh validate --all`, and the legacy `exporters/` tree is removed.

**Tech Stack:** Python 3.12, Pydantic v2, Click, PyYAML (all already deps). Tests:
pytest + `click.testing.CliRunner`. Container-first via `./dev run`.

**Scope note — dashboard producer:** SP1 scope (spec §7.2) includes a `dashboard`
producer, but the agreed increment order assigns it no slot. It is **not** built
here. The CLI is written kind-agnostically and stress-tested against a fake
second kind (registered in tests) so the abstraction is proven without the real
producer. The dashboard producer remains to be slotted (SP1.5b or folded into
SP1.6).

```python
"""Assemble CatalogEntry objects into the kind-agnostic catalog.json (spec §10)."""

from __future__ import annotations

import json
from collections.abc import Iterable, Mapping
from datetime import UTC, datetime
from pathlib import Path

from pydantic import ValidationError

from forge.domain.catalog import Catalog, CatalogEntry
from forge.domain.errors import ForgeError


def _utc_now() -> str:
    return datetime.now(UTC).strftime("%Y-%m-%dT%H:%M:%SZ")


def build_catalog(
    entries: Iterable[CatalogEntry],
    *,
    previous: Catalog | None = None,
    generated_at: str | None = None,
) -> Catalog:
    """Assemble entries into a Catalog, computing new/updated vs ``previous``."""
    prior: Mapping[tuple[str, str], str] = (
        {(e.kind, e.name): e.version for e in previous.items} if previous else {}
    )
    items: list[CatalogEntry] = []
    for entry in entries:
        key = (entry.kind, entry.name)
        old_version = prior.get(key)
        items.append(
            entry.model_copy(
                update={
                    "new": old_version is None,
                    "updated": old_version is not None and old_version != entry.version,
                }
            )
        )
    return Catalog(generated_at=generated_at or _utc_now(), items=items)


def load_catalog(path: Path) -> Catalog | None:
    """Load a previous catalog.json; return None if the file does not exist."""
    if not path.is_file():
        return None
    try:
        return Catalog.model_validate_json(path.read_text(encoding="utf-8"))
    except (OSError, ValidationError, ValueError) as exc:
        raise ForgeError(f"cannot read catalog {path}: {exc}") from exc


def write_catalog(catalog: Catalog, path: Path) -> Path:
    """Serialize a Catalog to indented JSON at ``path``."""
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        json.dumps(catalog.model_dump(mode="json"), indent=2, sort_keys=False) + "\n",
        encoding="utf-8",
    )
    return path
```

```python
"""CLI helpers: catalogue-root resolution and manifest iteration."""

from __future__ import annotations

import os
from collections.abc import Iterator
from pathlib import Path

_KIND_DIRS = ("exporters", "dashboards")
_MANIFEST = "manifest.yaml"


def resolve_catalog_root(option: str | None) -> Path:
    if option:
        return Path(option)
    env = os.environ.get("FORGE_CATALOG_ROOT")
    return Path(env) if env else Path("catalog")


def iter_manifest_paths(root: Path, *, kind: str | None = None) -> Iterator[Path]:
    dirs = (f"{kind}s",) if kind else _KIND_DIRS
    for kind_dir in dirs:
        base = root / kind_dir
        if not base.is_dir():
            continue
        for item in sorted(base.iterdir()):
            manifest = item / _MANIFEST
            if manifest.is_file():
                yield manifest
```
